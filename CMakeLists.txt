cmake_minimum_required(VERSION 3.22)
project(LibrEQ VERSION 0.1.0 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(LibrEQ COMPANY_NAME "LibrEQ Community" BUNDLE_ID org.libreq.audio.libreq PLUGIN_MANUFACTURER_CODE LiEQ PLUGIN_CODE LiEQ FORMATS VST3 PRODUCT_NAME "LibrEQ" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD TRUE)
juce_generate_juce_header(LibrEQ)
target_sources(LibrEQ PRIVATE src/Plugin/PluginProcessor.cpp src/Plugin/PluginEditor.cpp src/DSP/Equalizer.cpp)
target_include_directories(LibrEQ PRIVATE src)
target_compile_features(LibrEQ PRIVATE cxx_std_20)
target_compile_definitions(LibrEQ PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(LibrEQ PRIVATE juce::juce_audio_utils juce::juce_dsp)
add_executable(LibrEQSmokeTest tests/SmokeTest.cpp src/DSP/Equalizer.cpp)
target_include_directories(LibrEQSmokeTest PRIVATE src)
target_compile_features(LibrEQSmokeTest PRIVATE cxx_std_20)
