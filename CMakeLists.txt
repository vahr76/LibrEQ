cmake_minimum_required(VERSION 3.22)
project(ArieloEQ VERSION 0.1.0 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(ArieloEQ COMPANY_NAME "Arielo Audio" BUNDLE_ID com.arielo.audio.arieloeq PLUGIN_MANUFACTURER_CODE ArEQ PLUGIN_CODE ArEQ FORMATS VST3 PRODUCT_NAME "Arielo EQ" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD TRUE)
target_sources(ArieloEQ PRIVATE src/Plugin/PluginProcessor.cpp src/Plugin/PluginEditor.cpp src/DSP/Equalizer.cpp)
target_include_directories(ArieloEQ PRIVATE src)
target_compile_features(ArieloEQ PRIVATE cxx_std_20)
target_compile_definitions(ArieloEQ PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(ArieloEQ PRIVATE juce::juce_audio_utils juce::juce_dsp)
add_executable(ArieloEQSmokeTest tests/SmokeTest.cpp src/DSP/Equalizer.cpp)
target_include_directories(ArieloEQSmokeTest PRIVATE src)
target_compile_features(ArieloEQSmokeTest PRIVATE cxx_std_20)
